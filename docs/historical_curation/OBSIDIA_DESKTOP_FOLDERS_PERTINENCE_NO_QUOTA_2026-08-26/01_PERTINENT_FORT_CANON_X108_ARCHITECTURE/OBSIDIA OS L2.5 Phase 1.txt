        // ============================================================
        // OS L2.5 IMPLEMENTATION — CANONICAL
        // ============================================================

        const IR_ALPHABET = {
            VALUE: true, STATE: true, READ: true, WRITE: true,
            FLOW: true, COND: true, LOOP: true, CALL: true,
            RETURN: true, EVENT: true, TIME: true, ERROR: true
        };

        class OSL25Error extends Error {
            constructor(rule, message) {
                super(`[${rule}] ${message}`);
                this.rule = rule;
            }
        }

        function OS_L25_validate(irSequence) {
            if (!Array.isArray(irSequence)) {
                throw new OSL25Error("R10", "IR sequence must be a list");
            }

            for (let instr of irSequence) {
                if (!Array.isArray(instr) || instr.length < 1) {
                    throw new OSL25Error("R10", "Instruction must be a non-empty tuple");
                }

                const symbol = instr[0];

                if (!IR_ALPHABET[symbol]) {
                    throw new OSL25Error("R10", `Symbol '${symbol}' not in IR alphabet`);
                }

                if (symbol === "READ" && instr.length !== 2) {
                    throw new OSL25Error("R10", "READ expects (READ, state)");
                }
                if (symbol === "WRITE" && instr.length !== 3) {
                    throw new OSL25Error("R10", "WRITE expects (WRITE, state, value)");
                }
                if (symbol === "TIME" && instr.length !== 2) {
                    throw new OSL25Error("R10", "TIME expects (TIME, t)");
                }
            }

            return true;
        }

        function OS_L25_execute(irSequence, stateStore) {
            OS_L25_validate(irSequence);

            let lastTime = null;

            for (let instr of irSequence) {
                const symbol = instr[0];

                if (symbol === "READ") {
                    const state = instr[1];
                    if (!(state in stateStore)) {
                        throw new OSL25Error("R1", `State '${state}' does not exist`);
                    }
                    return stateStore[state];

                } else if (symbol === "WRITE") {
                    const state = instr[1];
                    const value = instr[2];
                    if (!(state in stateStore)) {
                        throw new OSL25Error("R1", `WRITE on non-existing state '${state}'`);
                    }
                    stateStore[state] = value;

                } else if (symbol === "TIME") {
                    const t = instr[1];
                    if (typeof t !== "number") {
                        throw new OSL25Error("R6", "TIME must be numeric");
                    }
                    if (lastTime !== null && t < lastTime) {
                        throw new OSL25Error("R6", "TIME must be monotonic");
                    }
                    lastTime = t;

                } else if (symbol === "ERROR") {
                    throw new OSL25Error("R7", "Explicit ERROR instruction");

                } else if (["FLOW", "COND", "LOOP", "CALL", "RETURN", "EVENT", "VALUE", "STATE"].includes(symbol)) {
                    // Allowed but inert in Phase 1
                    continue;
                } else {
                    throw new OSL25Error("R10", `Unhandled symbol ${symbol}`);
                }
            }

            return null;
        }

        class AgentA1 {
            constructor() {
                this.stateStore = {};
            }

            initState(name, value) {
                if (name in this.stateStore) {
                    throw new OSL25Error("R1", `State '${name}' already exists`);
                }
                this.stateStore[name] = value;
            }

            memoryContinuite(irSequence) {
                return OS_L25_execute(irSequence, this.stateStore);
            }
        }

        function runTests() {
            clearLog();
            log("🚀 Starting test suite...", "info");

            const agent = new AgentA1();
            agent.initState("x", 0);
            log("[SETUP] AgentA1 initialized, state_store.x = 0", "success");

            let passed = 0;
            let failed = 0;

            for (let test of testCases) {
                try {
                    const result = agent.memoryContinuite(test.ir);
                    if (test.shouldPass) {
                        if (result === test.expected) {
                            log(`✓ ${test.name}: PASS (result=${result})`, "success");
                            passed++;
                        } else {
                            log(`✗ ${test.name}: FAIL (expected ${test.expected}, got ${result})`, "error");
                            failed++;
                        }
                    } else {
                        log(`✗ ${test.name}: FAIL (should have thrown error)`, "error");
                        failed++;
                    }
                } catch (e) {
                    if (!test.shouldPass && e.rule === test.expectedRule) {
                        log(`✓ ${test.name}: PASS (caught ${e.rule})`, "success");
                        passed++;
                    } else {
                        log(`✗ ${test.name}: ${e.message}`, "error");
                        failed++;
                    }
                }
            }

            log(`\n📊 Results: ${passed}/${testCases.length} passed, ${failed} failed`, passed === testCases.length ? "success" : "error");
            log("✅ Test suite complete", "info");
        }
