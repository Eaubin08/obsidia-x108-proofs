import Std

/-
  OBSIDIA — Formal Core V18.9
  Lean 4.28 — Phase 11.2 (core theorems)

  Domain: Rat (no Mathlib)
-/

namespace Obsidia

/-- OS2 engine metrics (Rat domain) -/
structure Metrics where
  T_mean  : Rat
  H_score : Rat
  A_score : Rat
  S       : Rat

/-- Binary decision -/
inductive Decision
  | HOLD
  | ACT
  deriving Repr, DecidableEq

/-- Decision rule: ACT iff θ ≤ S -/
def decision (m : Metrics) (theta : Rat) : Decision :=
  if theta ≤ m.S then Decision.ACT else Decision.HOLD

/-- Characterization: ACT iff θ ≤ S. -/
theorem decision_eq_ACT_iff (m : Metrics) (theta : Rat) :
    decision m theta = Decision.ACT ↔ theta ≤ m.S := by
  unfold decision
  by_cases h : theta ≤ m.S
  · simp [h]
  · simp [h]

/-- Characterization: HOLD iff ¬(θ ≤ S). -/
theorem decision_eq_HOLD_iff (m : Metrics) (theta : Rat) :
    decision m theta = Decision.HOLD ↔ ¬ theta ≤ m.S := by
  unfold decision
  by_cases h : theta ≤ m.S
  · simp [h]
  · simp [h]

/-- D1 — Determinism (pure function). -/
theorem D1_determinism (m : Metrics) (theta : Rat) :
    decision m theta = decision m theta := rfl

/-- G1 — ACT if θ ≤ S. -/
theorem G1_act_above_threshold (m : Metrics) (theta : Rat)
    (h : theta ≤ m.S) :
    decision m theta = Decision.ACT := (decision_eq_ACT_iff m theta).2 h

/-- E2 — HOLD if ¬(θ ≤ S). -/
theorem E2_no_act_below_threshold (m : Metrics) (theta : Rat)
    (h : ¬ theta ≤ m.S) :
    decision m theta = Decision.HOLD := (decision_eq_HOLD_iff m theta).2 h

/-- G2 — Boundary inclusive: S = θ → ACT. -/
theorem G2_boundary_inclusive (m : Metrics) (theta : Rat)
    (h : m.S = theta) :
    decision m theta = Decision.ACT := by
  have : theta ≤ m.S := by
    simpa [h] using (Rat.le_refl : theta ≤ theta)
  exact G1_act_above_threshold m theta this

/-- G3 — Monotonicity: if θ ≤ S₁ and S₁ ≤ S₂ then decision₂ = ACT. -/
theorem G3_monotonicity (m₁ m₂ : Metrics) (theta : Rat)
    (h1 : theta ≤ m₁.S) (h2 : m₁.S ≤ m₂.S) :
    decision m₂ theta = Decision.ACT := by
  have : theta ≤ m₂.S := Rat.le_trans h1 h2
  exact G1_act_above_threshold m₂ theta this

/-
  Phase 11.3 — Institutional decision layer preparation

  We introduce a tri-state decision for institutional stacks:
  BLOCK > HOLD > ACT

  In OS2 core, the computed decision is binary (HOLD/ACT).
  We lift it into Decision3, and prove it never yields BLOCK.
-/

/-- Tri-state institutional decision (for future consensus / PBFT / refusal). -/
inductive Decision3
  | BLOCK
  | HOLD
  | ACT
  deriving Repr, DecidableEq

/-- Lift binary Decision into Decision3 (no BLOCK at OS2 core). -/
def liftDecision (d : Decision) : Decision3 :=
  match d with
  | Decision.HOLD => Decision3.HOLD
  | Decision.ACT  => Decision3.ACT

/-- Institutional decision rule: lifted OS2 decision (no BLOCK here). -/
def decision3 (m : Metrics) (theta : Rat) : Decision3 :=
  liftDecision (decision m theta)

/-- L11.3 — Core never outputs BLOCK. -/
theorem L11_3_no_block (m : Metrics) (theta : Rat) :
    decision3 m theta ≠ Decision3.BLOCK := by
  unfold decision3 liftDecision
  cases h : decision m theta <;> intro hb <;> cases hb

/-- L11.3 — If θ ≤ S then decision3 = ACT. -/
theorem L11_3_act (m : Metrics) (theta : Rat) (h : theta ≤ m.S) :
    decision3 m theta = Decision3.ACT := by
  unfold decision3 liftDecision
  rw [G1_act_above_threshold m theta h]

/-- L11.3 — If ¬(θ ≤ S) then decision3 = HOLD. -/
theorem L11_3_hold (m : Metrics) (theta : Rat) (h : ¬ theta ≤ m.S) :
    decision3 m theta = Decision3.HOLD := by
  unfold decision3 liftDecision
  rw [E2_no_act_below_threshold m theta h]

/-- A kernel decision cannot be both ACT and HOLD. -/
theorem decision_not_both (m : Metrics) (theta : Rat) :
    ¬ (decision m theta = Decision.ACT ∧ decision m theta = Decision.HOLD) := by
  intro h
  rcases h with ⟨h1, h2⟩
  rw [h1] at h2
  cases h2

end Obsidia

import Obsidia.Basic

/-
  OBSIDIA — Phase 20b : Temporal Gate Formalization (X-108)

  This module formalizes a minimal temporal gate:

    - Kernel decision remains binary: Decision = {HOLD, ACT}
    - For irreversible intents (irr = true),
      ACT is forbidden before elapsed ≥ τ.

  Time domain: Int (allows negative elapsed for clock skew robustness).
  No Mathlib.

  This module is intentionally minimal and composable.
-/

namespace Obsidia

/-- Temporal input for X-108 gating. -/
structure TInput where
  metrics   : Metrics
  theta     : Rat
  irr       : Bool
  createdAt : Int
  now       : Int

/-- Elapsed time, allowing negative values (clock skew). -/
def elapsed (i : TInput) : Int :=
  i.now - i.createdAt

/-- Gate threshold τ (seconds). Caller chooses τ ≥ 0. -/
abbrev Tau := Int

/-- X-108 temporal gate predicate. -/
def beforeTau (τ : Tau) (i : TInput) : Bool :=
  i.irr && (elapsed i < τ)

/-- Temporal kernel decision. -/
def decideX108 (τ : Tau) (i : TInput) : Decision :=
  if beforeTau τ i then Decision.HOLD
  else decision i.metrics i.theta

/-- Lift to institutional tri-state (still no BLOCK at kernel). -/
def decide3X108 (τ : Tau) (i : TInput) : Decision3 :=
  liftDecision (decideX108 τ i)

/-- X108-1: No ACT before τ for irreversible intents. -/
theorem X108_no_act_before_tau (τ : Tau) (i : TInput)
    (hIrr : i.irr = true)
    (h : elapsed i < τ) :
    decideX108 τ i = Decision.HOLD := by
  unfold decideX108 beforeTau
  have hb : (i.irr && (elapsed i < τ)) = true := by
    simp [hIrr, h]
  simp [hb]

/-- X108-2: Negative elapsed implies HOLD when τ ≥ 0. -/
theorem X108_skew_safe (τ : Tau) (i : TInput)
    (hIrr : i.irr = true)
    (hneg : elapsed i < 0)
    (hTau : 0 ≤ τ) :
    decideX108 τ i = Decision.HOLD := by
  have hlt : elapsed i < τ := Int.lt_of_lt_of_le hneg hTau
  exact X108_no_act_before_tau τ i hIrr hlt

/-- X108-3: After τ, kernel decision equals base decision (gate inactive). -/
theorem X108_after_tau_equals_base (τ : Tau) (i : TInput)
    (h : ¬ ((i.irr && (elapsed i < τ)) = true)) :
    decideX108 τ i = decision i.metrics i.theta := by
  unfold decideX108 beforeTau
  simp [h]

/-- X108-4: Institutional lift still never produces BLOCK. -/
theorem X108_kernel_never_blocks (τ : Tau) (i : TInput) :
    decide3X108 τ i ≠ Decision3.BLOCK := by
  unfold decide3X108
  cases decideX108 τ i <;> simp [liftDecision]

/-- X108-5: If intent is reversible, X-108 does not alter the base decision. -/
theorem X108_reversible_equals_base (τ : Tau) (i : TInput)
    (hIrr : i.irr = false) :
    decideX108 τ i = decision i.metrics i.theta := by
  unfold decideX108 beforeTau
  simp [hIrr]

/-- X108-6: For irreversible intents, once elapsed ≥ τ, X-108 equals base decision. -/
theorem X108_irreversible_after_tau_equals_base (τ : Tau) (i : TInput)
    (hIrr : i.irr = true)
    (hTau : τ ≤ elapsed i) :
    decideX108 τ i = decision i.metrics i.theta := by
  unfold decideX108 beforeTau
  have hNotLt : ¬ (elapsed i < τ) := by
    exact Int.not_lt.mpr hTau
  simp [hIrr, hNotLt]

end Obsidia

namespace Obsidia.Refinement

/-- Refinement relation between kernel Decision and institutional Decision3. -/
def R_decision : Decision → Decision3 → Prop
| Decision.HOLD, Decision3.HOLD => True
| Decision.ACT,  Decision3.ACT  => True
| _, _ => False

/-- liftDecision refines kernel decision into institutional decision. -/
theorem lift_refines (d : Decision) :
    R_decision d (liftDecision d) := by
  cases d <;> simp [R_decision, liftDecision]

/-- X-108 decision3 is always a lift of a kernel decision. -/
theorem x108_is_lift (τ : Tau) (i : TInput) :
    decide3X108 τ i = liftDecision (decideX108 τ i) := by
  rfl

/-- Conformance lemma: X-108 never introduces BLOCK (institutional view). -/
theorem x108_never_blocks (τ : Tau) (i : TInput) :
    decide3X108 τ i ≠ Decision3.BLOCK := by
  exact Obsidia.X108_kernel_never_blocks τ i

/-- Any institutional decision refining a kernel decision cannot be BLOCK. -/
theorem refined_not_block (d : Decision) (d3 : Decision3)
    (h : R_decision d d3) :
    d3 ≠ Decision3.BLOCK := by
  cases d <;> cases d3 <;> simp [R_decision] at h ⊢

end Obsidia.Refinement

/-
  OBSIDIA — Phase 11.5
  Consensus 3/4 (4 nodes) over Decision3, without Mathlib.
-/

import Obsidia.Basic

namespace Obsidia

/-- Count occurrences of a value in a list (no Mathlib). -/
def countDec (d : Decision3) (xs : List Decision3) : Nat :=
  xs.foldl (fun acc x => if x = d then acc + 1 else acc) 0

/-- Consensus aggregator for exactly 4 nodes (supermajority 3/4). -/
def aggregate4 (d1 d2 d3 d4 : Decision3) : Decision3 :=
  let xs : List Decision3 := [d1, d2, d3, d4]
  if 3 ≤ countDec Decision3.ACT xs then
    Decision3.ACT
  else if 3 ≤ countDec Decision3.HOLD xs then
    Decision3.HOLD
  else if 3 ≤ countDec Decision3.BLOCK xs then
    Decision3.BLOCK
  else
    Decision3.BLOCK

/-- If the ACT supermajority condition holds, aggregate4 = ACT. -/
theorem aggregate4_act
  (d1 d2 d3 d4 : Decision3)
  (h : 3 ≤ countDec Decision3.ACT [d1, d2, d3, d4]) :
  aggregate4 d1 d2 d3 d4 = Decision3.ACT := by
  unfold aggregate4
  simp only [countDec, List.foldl] at *
  cases d1 <;> cases d2 <;> cases d3 <;> cases d4 <;>
    simp_all (config := { decide := true })

/-- If ACT does not supermajority, but HOLD does, aggregate4 = HOLD. -/
theorem aggregate4_hold
  (d1 d2 d3 d4 : Decision3)
  (hAct : ¬ (3 ≤ countDec Decision3.ACT [d1, d2, d3, d4]))
  (hHold : 3 ≤ countDec Decision3.HOLD [d1, d2, d3, d4]) :
  aggregate4 d1 d2 d3 d4 = Decision3.HOLD := by
  unfold aggregate4
  simp only [countDec, List.foldl] at *
  cases d1 <;> cases d2 <;> cases d3 <;> cases d4 <;>
    simp_all (config := { decide := true })

/-- If neither ACT nor HOLD supermajority, but BLOCK does, aggregate4 = BLOCK. -/
theorem aggregate4_block_by_supermajority
  (d1 d2 d3 d4 : Decision3)
  (hAct : ¬ (3 ≤ countDec Decision3.ACT [d1, d2, d3, d4]))
  (hHold : ¬ (3 ≤ countDec Decision3.HOLD [d1, d2, d3, d4]))
  (hBlock : 3 ≤ countDec Decision3.BLOCK [d1, d2, d3, d4]) :
  aggregate4 d1 d2 d3 d4 = Decision3.BLOCK := by
  unfold aggregate4
  simp only [countDec, List.foldl] at *
  cases d1 <;> cases d2 <;> cases d3 <;> cases d4 <;>
    simp_all (config := { decide := true })

/-- Fail-closed: if no decision reaches 3/4, output is BLOCK. -/
theorem aggregate4_fail_closed
  (d1 d2 d3 d4 : Decision3)
  (hAct : ¬ (3 ≤ countDec Decision3.ACT [d1, d2, d3, d4]))
  (hHold : ¬ (3 ≤ countDec Decision3.HOLD [d1, d2, d3, d4]))
  (hBlock : ¬ (3 ≤ countDec Decision3.BLOCK [d1, d2, d3, d4])) :
  aggregate4 d1 d2 d3 d4 = Decision3.BLOCK := by
  unfold aggregate4
  simp only [countDec, List.foldl] at *
  cases d1 <;> cases d2 <;> cases d3 <;> cases d4 <;>
    simp_all (config := { decide := true })

/-- Unanimity: if all 4 nodes agree, aggregate4 returns that decision. -/
theorem aggregate4_unanimous (d : Decision3) :
  aggregate4 d d d d = d := by
  cases d <;> simp [aggregate4, countDec, List.foldl]

/-- If three nodes vote ACT, aggregate4 returns ACT regardless of the fourth. -/
theorem aggregate4_three_act (d : Decision3) :
  aggregate4 Decision3.ACT Decision3.ACT Decision3.ACT d = Decision3.ACT := by
  cases d <;> simp [aggregate4, countDec, List.foldl]

/-- For 4 nodes and threshold 3, two distinct decisions cannot both have supermajority. -/
theorem no_two_distinct_supermajorities_4
  (a b : Decision3) (d1 d2 d3 d4 : Decision3)
  (hab : a ≠ b)
  (ha : 3 ≤ countDec a [d1, d2, d3, d4])
  (hb : 3 ≤ countDec b [d1, d2, d3, d4]) :
  False := by
  cases a <;> cases b <;> cases d1 <;> cases d2 <;> cases d3 <;> cases d4 <;>
    simp_all (config := { decide := true })

end Obsidia

import Obsidia.Basic
import Obsidia.Sensitivity

namespace Obsidia.SystemModel

/-- An audit record links an input to the produced kernel decision. -/
structure AuditRecord where
  metrics : Obsidia.Metrics
  theta   : Rat
  result  : Obsidia.Decision

/-- Full system state. -/
structure State where
  repo     : Obsidia.Sensitivity.Repo
  auditLog : List AuditRecord

/-- System input. -/
structure Input where
  metrics : Obsidia.Metrics
  theta   : Rat

/-- Kernel decision function. -/
def decide (i : Input) : Obsidia.Decision :=
  Obsidia.decision i.metrics i.theta

/-- Lift to institutional decision. -/
def decideInstitutional (i : Input) : Obsidia.Decision3 :=
  Obsidia.liftDecision (decide i)

/-- The kernel never produces BLOCK at the institutional level. -/
theorem P17_KernelNeverBlocks (i : Input) :
    decideInstitutional i ≠ Obsidia.Decision3.BLOCK :=
  Obsidia.L11_3_no_block i.metrics i.theta

/-- Repository seal. -/
def sealRepo (r : Obsidia.Sensitivity.Repo) : Obsidia.Hash :=
  Obsidia.Sensitivity.merkleRoot r

/-- System transition function. -/
def transition (s : State) (i : Input) : Obsidia.Decision × State :=
  let d    := decide i
  let entry := AuditRecord.mk i.metrics i.theta d
  let s'   := State.mk s.repo (s.auditLog ++ [entry])
  (d, s')

/-- Determinism. -/
theorem P17_Determinism (s : State) (i : Input) :
    transition s i = transition s i := rfl

/-- Seal sensitivity. -/
theorem P17_SealSensitive
    (r r' : Obsidia.Sensitivity.Repo)
    (h : r.leaves ≠ r'.leaves) :
    sealRepo r ≠ sealRepo r' := by
  unfold sealRepo
  exact Obsidia.Sensitivity.merkleRoot_change_if_leaf_change r r' h

/-- Every transition adds exactly one audit record. -/
theorem P17_AuditGrowth (s : State) (i : Input) :
    (transition s i).2.auditLog.length = s.auditLog.length + 1 := by
  unfold transition
  simp [List.length_append]

/-- The appended audit record exactly matches the input and computed decision. -/
theorem P17_AuditLastIsComputed (s : State) (i : Input) :
    (transition s i).2.auditLog =
      s.auditLog ++ [AuditRecord.mk i.metrics i.theta (decide i)] := by
  unfold transition
  rfl

/-- The decision returned by transition is exactly the kernel decision on input i. -/
theorem P17_TransitionFstIsDecide (s : State) (i : Input) :
    (transition s i).1 = decide i := by
  unfold transition
  rfl

end Obsidia.SystemModel

namespace Obsidia

/-- Concrete hash type (closed toy model). -/
inductive Hash where
  | leaf : Nat → Hash
  | node : Hash → Hash → Hash
  deriving Repr, DecidableEq

/-- Concrete binary combine. -/
def H (a b : Hash) : Hash :=
  Hash.node a b

/-- Left injectivity of H. -/
theorem H_injective_left
  {a b c : Hash} (h : H a b = H c b) : a = c := by
  unfold H at h
  cases h
  rfl

/-- Right injectivity of H. -/
theorem H_injective_right
  {a b c : Hash} (h : H a b = H a c) : b = c := by
  unfold H at h
  cases h
  rfl

/-- Neutral hash used as initial accumulator. -/
def neutralHash : Hash :=
  Hash.leaf 0

/-- Decode a fold-built hash back into its leaf list, when possible. -/
def decodeFold : Hash → Option (List Hash)
  | Hash.leaf 0 => some []
  | Hash.leaf _ => none
  | Hash.node left right =>
      match decodeFold left with
      | some xs => some (xs ++ [right])
      | none => none

/-- If the accumulator already decodes to `prefList`, folding `xs` appends exactly `xs`. -/
theorem decodeFold_foldl_of_decodable :
    ∀ (xs : List Hash) (acc : Hash) (prefList : List Hash),
      decodeFold acc = some prefList →
      decodeFold (List.foldl H acc xs) = some (prefList ++ xs) := by
  intro xs
  induction xs with
  | nil =>
      intro acc prefList hdec
      simpa [List.foldl, hdec]
  | cons x xs ih =>
      intro acc prefList hdec
      have hstep : decodeFold (H acc x) = some (prefList ++ [x]) := by
        unfold H
        simp [decodeFold, hdec]
      simpa [List.foldl, List.append_assoc] using
        ih (H acc x) (prefList ++ [x]) hstep

/-- Decoding a fold from neutralHash returns exactly the original list. -/
theorem decodeFold_foldl_neutral (xs : List Hash) :
    decodeFold (List.foldl H neutralHash xs) = some xs := by
  have h0 : decodeFold neutralHash = some [] := by
    rfl
  simpa [neutralHash] using
    decodeFold_foldl_of_decodable xs neutralHash [] h0

/-- Injectivity of the fold when started from neutralHash. -/
theorem foldl_H_injective
    (xs ys : List Hash)
    (h : List.foldl H neutralHash xs = List.foldl H neutralHash ys) :
    xs = ys := by
  have hx : decodeFold (List.foldl H neutralHash xs) = some xs :=
    decodeFold_foldl_neutral xs
  have hy : decodeFold (List.foldl H neutralHash ys) = some ys :=
    decodeFold_foldl_neutral ys
  rw [h] at hx
  rw [hy] at hx
  exact Option.some.inj hx.symm

end Obsidia

/-- Simple Merkle over 2 leaves. -/
def merkle2 (a b : Hash) : Hash :=
  H a b

/-- Left mutation changes the 2-leaf Merkle root. -/
theorem merkle2_left_mutation
  (a a' b : Hash)
  (h : a ≠ a') :
  merkle2 a b ≠ merkle2 a' b := by
  unfold merkle2
  intro h_eq
  apply h
  exact H_injective_left h_eq

/-- Right mutation changes the 2-leaf Merkle root. -/
theorem merkle2_right_mutation
  (a b b' : Hash)
  (h : b ≠ b') :
  merkle2 a b ≠ merkle2 a b' := by
  unfold merkle2
  intro h_eq
  apply h
  exact H_injective_right h_eq

end Obsidia

import Obsidia.Merkle
import Obsidia.CryptoAssumptions

namespace Obsidia.Sensitivity

/-- A repository is an ordered list of file hashes. -/
structure Repo where
  leaves : List Obsidia.Hash

/-- Replace the hash at position i in the Repo. -/
def updateFile (i : Nat) (h : Obsidia.Hash) (r : Repo) : Repo :=
  { leaves := r.leaves.set i h }

/-- Combine a list of hashes into a Merkle-style root by left fold. -/
def merkleRoot (r : Repo) : Obsidia.Hash :=
  r.leaves.foldl Obsidia.H Obsidia.neutralHash

/-- globalSeal combines the manifest hash and the Merkle root. -/
def globalSeal (manifest root : Obsidia.Hash) : Obsidia.Hash :=
  Obsidia.merkle2 manifest root

/-- If the leaves of two repos differ, their Merkle roots differ. -/
theorem merkleRoot_change_if_leaf_change
    (r r' : Repo)
    (h : r.leaves ≠ r'.leaves) :
    merkleRoot r ≠ merkleRoot r' := by
  unfold merkleRoot
  intro heq
  apply h
  exact Obsidia.foldl_H_injective r.leaves r'.leaves heq

/-- If two Merkle roots differ, the corresponding global seals differ. -/
theorem globalSeal_change_if_root_change
    (manifest root root' : Obsidia.Hash)
    (h : root ≠ root') :
    globalSeal manifest root ≠ globalSeal manifest root' := by
  unfold globalSeal
  exact Obsidia.merkle2_right_mutation manifest root root' h

/-- Strong global immutability theorem. -/
theorem P15_Immutability_Strong
    (manifest : Obsidia.Hash)
    (repo repo' : Repo)
    (h : repo ≠ repo') :
    globalSeal manifest (merkleRoot repo)
    ≠
    globalSeal manifest (merkleRoot repo') := by
  have hleaves : repo.leaves ≠ repo'.leaves := by
    intro heq
    apply h
    cases repo
    cases repo'
    simp_all
  have hroot : merkleRoot repo ≠ merkleRoot repo' :=
    merkleRoot_change_if_leaf_change repo repo' hleaves
  exact globalSeal_change_if_root_change manifest (merkleRoot repo) (merkleRoot repo') hroot

end Obsidia.Sensitivity

import Obsidia.Merkle

/-- Retro-compatible alias. -/
theorem fileHash_inj (f g : File) (h : fileHash f = fileHash g) : f = g :=
  SealAssumptions.fileHash_inj f g h

/-- Legacy root hash via combine over file hashes. -/
def rootHash (files : List File) : Obsidia.Hash :=
  combine (files.map fileHash)

/-- Global seal combines manifest hash and root hash. -/
def globalSeal (mHash rHash : Obsidia.Hash) : Obsidia.Hash :=
  Obsidia.merkle2 mHash rHash

/-- If mapped file hashes differ, the global seal differs. -/
theorem P13_Immutability
    (manifest : Obsidia.Hash) (files files2 : List File)
    (hmap : files.map fileHash ≠ files2.map fileHash) :
    globalSeal manifest (rootHash files) ≠ globalSeal manifest (rootHash files2) := by
  have hroot : rootHash files ≠ rootHash files2 := by
    unfold rootHash
    intro heq
    exact hmap (combine_inj _ _ heq)
  unfold globalSeal
  exact Obsidia.merkle2_right_mutation manifest (rootHash files) (rootHash files2) hroot

end Obsidia.Seal
